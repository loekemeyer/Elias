# ---- 1. Bloquear los scripts de mantenimiento de storage -------------
# Por nombre exacto. NO se puede bloquear *.js en bloque: el frontend del
# sitio es JavaScript (script.js, admin.js, historial.js, version.js,
# admin-supercot.js, admin-excel-krikos.js...). Un deny sobre .js
# rompe el sitio entero.
<FilesMatch "^(check-non-webp|compress-75kb|download-all-images|optimize-images|recreate-bucket|rename-images|storage-audit|storage-size|supabase-admin-client)\.js$">
  # Apache 2.4
  Require all denied
  # Apache 2.2 (descomentar solo si el host es 2.2)
  # Order allow,deny
  # Deny from all
</FilesMatch>

# ---- 2. Bloquear artefactos que nunca deben servirse ------------------
<FilesMatch "\.(sql|md|log|bak|old|zip|ps1)$">
  Require all denied
</FilesMatch>

# ---- 3. Bloquear dotfiles y archivos de credenciales ------------------
<FilesMatch "^\.">
  Require all denied
</FilesMatch>
<FilesMatch "^(package\.json|package-lock\.json)$">
  Require all denied
</FilesMatch>

# ---- 4. Bloquear directorios de repo y planificacion -----------------
RedirectMatch 404 ^/\.git(/|$)
RedirectMatch 404 ^/\.planning(/|$)
RedirectMatch 404 ^/\.locks(/|$)
RedirectMatch 404 ^/node_modules(/|$)
RedirectMatch 404 ^/backup-images(/|$)

# ---- 5. No listar directorios ----------------------------------------
Options -Indexes

# ---- 6. Redirect a HTTPS ---------------------------------------------
# Va DESPUES de los denies a proposito. Si fuera antes, un pedido por
# http:// a un script bloqueado devolveria 301 en vez de 403, confirmando
# que el archivo existe.
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{HTTPS} off
  RewriteRule ^(.*)$ https://%{HTTP_HOST}/$1 [R=301,L]
</IfModule>

# ---- 7. Headers de seguridad -----------------------------------------
<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>
